    function resizeNeural() {

      const rect =
        neuralCanvas.parentElement.getBoundingClientRect();

      const dpr =
        Math.min(
          window.devicePixelRatio || 1,
          2
        );

      neuralWidth =
        Math.max(1, rect.width);

      neuralHeight =
        Math.max(1, rect.height);

      neuralCanvas.width =
        neuralWidth * dpr;

      neuralCanvas.height =
        neuralHeight * dpr;

      neuralCanvas.style.width =
        `${neuralWidth}px`;

      neuralCanvas.style.height =
        `${neuralHeight}px`;

      neuralCtx.setTransform(
        dpr,
        0,
        0,
        dpr,
        0,
        0
      );

      neuralNodes =
        Array.from(
          { length: 36 },
          () => ({
            x:
              Math.random() *
              neuralWidth,

            y:
              Math.random() *
              neuralHeight,

            vx:
              (Math.random() - .5) *
              .25,

            vy:
              (Math.random() - .5) *
              .25,

            r:
              Math.random() * 2 +
              1
          })
        );
    }

    function animateNeural() {

      neuralCtx.clearRect(
        0,
        0,
        neuralWidth,
        neuralHeight
      );

      for (const node of neuralNodes) {

        node.x += node.vx;
        node.y += node.vy;

        if (
          node.x < 0 ||
          node.x > neuralWidth
        ) {
          node.vx *= -1;
        }

        if (
          node.y < 0 ||
          node.y > neuralHeight
        ) {
          node.vy *= -1;
        }
      }

      for (
        let i = 0;
        i < neuralNodes.length;
        i++
      ) {

        for (
          let j = i + 1;
          j < neuralNodes.length;
          j++
        ) {

          const a =
            neuralNodes[i];

          const b =
            neuralNodes[j];

          const dx =
            a.x - b.x;

          const dy =
            a.y - b.y;

          const distance =
            Math.sqrt(
              dx * dx +
              dy * dy
            );

          if (distance < 130) {

            const opacity =
              (1 -
                distance / 130) *
              .28;

            neuralCtx.beginPath();

            neuralCtx.moveTo(
              a.x,
              a.y
            );

            neuralCtx.lineTo(
              b.x,
              b.y
            );

            neuralCtx.strokeStyle =
              `rgba(0,229,255,${opacity})`;

            neuralCtx.lineWidth =
              .6;

            neuralCtx.stroke();
          }
        }
      }

      for (const node of neuralNodes) {

        neuralCtx.beginPath();

        neuralCtx.arc(
          node.x,
          node.y,
          node.r,
          0,
          Math.PI * 2
        );

        neuralCtx.fillStyle =
          "rgba(0,229,255,.75)";

        neuralCtx.shadowBlur =
          12;

        neuralCtx.shadowColor =
          "#00e5ff";

        neuralCtx.fill();

        neuralCtx.shadowBlur = 0;
      }

      requestAnimationFrame(
        animateNeural
      );
    }

    animateNeural();
